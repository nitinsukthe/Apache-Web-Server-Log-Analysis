##############################
# 2. Install Apache Web Server
##############################

sudo apt install apache2 -y

##############################
# 3. Start Apache Service
##############################

sudo systemctl start apache2

##############################
# 4. Enable Apache on Startup
##############################

sudo systemctl enable apache2

##############################
# 5. Verify Apache Status
##############################

sudo systemctl status apache2

http://localhost

# 9. Navigate to Apache Logs
##############################

cd /var/log/apache2

##############################
# 10. Verify Current Directory
##############################

pwd

##############################
# 11. List Available Log Files
##############################

ls -lh

##############################
# 12. View Apache Access Logs
##############################

sudo less access.log

# Exit less
q

##############################
# 13. Search for HTTP 404 Errors
##############################

grep " 404 " access.log

##############################
# 14. Count Requests by Source IP
##############################

awk '{print $1}' access.log | sort | uniq -c | sort -nr

##############################
# 15. Identify Most Requested URLs
##############################

awk '{print $7}' access.log | sort | uniq -c | sort -nr

##############################
# 16. Review Apache Error Logs
##############################

cat error.log

# Alternative
tail -20 error.log

##############################
# 17. Count Total 404 Errors
##############################

grep " 404 " access.log | wc -l

##############################
# 18. Create Security Findings File
##############################

nano findings.txt

##############################
# 19. View Findings File
##############################

cat findings.txt
